# OCR backends adicionales
# -----------------------------------------------------------------------------

# Tesseract ya viene en requirements.txt base — no repetir aquí.

# PaddleOCR PP-OCRv5 — recomendado; mejor soporte español, det+rec integrado.
# paddleocr>=3.0
# paddlepaddle>=3.0          # CPU puro
# paddlepaddle-gpu>=3.0      # GPU CUDA (alternativa, descomentar solo una)

# docTR — bueno para texto impreso, layout complejo.
# python-doctr[torch]>=0.8

# EasyOCR — ligero, buen fallback.
# easyocr>=1.7

# -----------------------------------------------------------------------------
# Detección de glifos alternativa
# -----------------------------------------------------------------------------

# CRAFT — detección a nivel de carácter (link_threshold bajo en CRAFTDetector).
# craft-text-detector>=0.4.3

# PaddleDetector usa el mismo paquete que PaddleOCR (ver arriba).

# -----------------------------------------------------------------------------
# Etiquetado automático de glifos (banco de tipografía)
# -----------------------------------------------------------------------------

# TrOCR de Microsoft — escrito a mano, descarga ~400 MB en primera ejecución.
# transformers>=4.40
# torch>=2.0                 # CPU; torch-cuda-XX para GPU

# -----------------------------------------------------------------------------
# Lectura de PDFs
# -----------------------------------------------------------------------------

# pdfplumber y pdf2image son REQUERIDOS y están en requirements.txt base.
# No repetir aquí.

# ocrmypdf — hacer PDFs escaneados "searchable" (opcional, pesado).
# Requiere: sudo dnf install ghostscript
# ocrmypdf>=16.0

# -----------------------------------------------------------------------------
# Preprocesamiento de imágenes "nivel escáner" (extractor de glifos)
# -----------------------------------------------------------------------------

# scikit-image — binarización de documento robusta (threshold_sauvola) y realce
# de contraste para letras tenues (equalize_adapthist / rescale_intensity).
# Mejora notablemente la extracción de fotos difíciles (sombra, bajo contraste).
# OPCIONAL: si no está instalado, el extractor degrada a Sauvola casero + CLAHE
# de OpenCV automáticamente (mismo resultado que antes, sin errores).
# scikit-image>=0.24
